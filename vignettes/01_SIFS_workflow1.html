X <- as.matrix(train_data[, setdiff(names(train_data), "target")])

dtrain <- xgb.DMatrix(data = X, label = y)

params <- list(
  objective = "binary:logistic",
  max_depth = 6,
  eta = 0.1,
  gamma = 0,
  colsample_bytree = 0.6,
  min_child_weight = 3,
  subsample = 1,
  eval_metric = "logloss"
)

model <- xgb.train(
  params = params,
  data = dtrain,
  nrounds = 200
)

xai_unify_xgboost <- function(xgb_model, data) {
  xgbtree <- xgboost::xgb.model.dt.tree(model = xgb_model)

  if (!("Gain" %in% colnames(xgbtree)) && "Quality" %in% colnames(xgbtree)) {
    xgbtree$Gain <- xgbtree$Quality
  }

  stopifnot(
    c(
      "Tree", "Node", "ID", "Feature", "Split",
      "Yes", "No", "Missing", "Gain", "Cover"
    ) %in% colnames(xgbtree)
  )

  xgbtree$Yes <- match(xgbtree$Yes, xgbtree$ID)
  xgbtree$No <- match(xgbtree$No, xgbtree$ID)
  xgbtree$Missing <- match(xgbtree$Missing, xgbtree$ID)
  xgbtree[is.na(xgbtree$Split), "Feature"] <- NA_character_
  xgbtree$Decision.type <- factor(rep("<=", nrow(xgbtree)), levels = c("<=", "<"))
  xgbtree$Decision.type[is.na(xgbtree$Feature)] <- NA_character_
  xgbtree <- xgbtree[, c(
    "Tree", "Node", "Feature", "Decision.type", "Split",
    "Yes", "No", "Missing", "Gain", "Cover"
  )]
  colnames(xgbtree) <- c(
    "Tree", "Node", "Feature", "Decision.type", "Split",
    "Yes", "No", "Missing", "Prediction", "Cover"
  )
  xgbtree$Prediction[!is.na(xgbtree$Feature)] <- NA_real_

  feature_names <- tryCatch(
    xgboost::getinfo(xgb_model, "feature_name"),
    error = function(e) NULL
  )
  if (is.null(feature_names) || !length(feature_names)) {
    feature_names <- colnames(data)
  }

  if (all(feature_names %in% colnames(data))) {
    data <- data[, feature_names, drop = FALSE]
  }

  unified_model <- list(
    model = as.data.frame(xgbtree),
    data = as.data.frame(data),
    feature_names = feature_names
  )
  class(unified_model) <- "model_unified"
  attr(unified_model, "missing_support") <- TRUE
  attr(unified_model, "model") <- "xgboost"
  unified_model
}
